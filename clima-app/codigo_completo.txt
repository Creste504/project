    DecimalPipe,
  ],
  providers: [DatePipe, DecimalPipe],
})
export class AlertasPage implements OnInit {
  // Coordenadas padrÃ£o: Recife, PE
  lat = -8.0543;
  lon = -34.8813;
  cidade = 'Recife, PE';

  previsaoDiaria: PrevisaoDiaria | null = null;
  carregando = true;
  erro = '';
  apenasAlertas = false;

  constructor(
    private climaService: ClimaService,
    private cdr: ChangeDetectorRef,
    private route: ActivatedRoute
  ) {}

  ngOnInit() {
    this.route.queryParams.subscribe((params) => {
      if (params['lat'] && params['lon']) {
        this.lat = parseFloat(params['lat']);
        this.lon = parseFloat(params['lon']);
        if (params['cidade']) {
          this.cidade = params['cidade'];
        }
      }
      this.carregarPrevisaoDiaria();
    });
  }

  carregarPrevisaoDiaria() {
    this.carregando = true;
    this.erro = '';
    this.climaService.getPrevisaoDiaria(this.lat, this.lon).subscribe({
      next: (data) => {
        this.previsaoDiaria = data;
        this.carregando = false;
        this.cdr.markForCheck();
      },
      error: (err) => {
        console.error('Erro ao buscar previsÃ£o diÃ¡ria:', err);
        this.erro = 'Falha ao carregar previsÃ£o diÃ¡ria da API.';
        this.carregando = false;
        this.cdr.markForCheck();
      },
    });
  }

  recarregar() {
    this.climaService.limparCache();
    this.carregarPrevisaoDiaria();
  }

  alternarFiltroAlertas() {
    this.apenasAlertas = !this.apenasAlertas;
    this.cdr.markForCheck();
  }

  temAlerta(i: number): boolean {
    if (!this.previsaoDiaria) return false;
    const d = this.previsaoDiaria.daily;
    return d.precipitation_sum[i] > 5 || d.uv_index_max[i] > 7 || d.weather_code[i] >= 80;
  }

  get diasIndices(): number[] {
    if (!this.previsaoDiaria) return [];
    const todos = this.previsaoDiaria.daily.time.map((_, i) => i);
    return this.apenasAlertas ? todos.filter((i) => this.temAlerta(i)) : todos;
  }
}

import { CommonModule, DecimalPipe } from '@angular/common';
import { ChangeDetectionStrategy, ChangeDetectorRef, Component, OnInit } from '@angular/core';
import { FormsModule } from '@angular/forms';
import {
  IonContent,

import { forkJoin } from 'rxjs';
import { ClimaService, ClimaAtual, ResultadoGeocodificacao } from '../services/clima.service';
import { DescricaoClimaPipe } from '../pipes/descricao-clima.pipe';
import { FaixaTemperaturaDirective } from '../directives/faixa-temperatura.directive';
import { DestaqueAlertaDirective } from '../directives/destaque-alerta.directive';

interface CidadeSalva {
  nome: string;
  lat: number;
  lon: number;
  clima?: ClimaAtual;
  carregando?: boolean;
  erro?: string;
}

@Component({
  selector: 'app-cidades',
  templateUrl: './cidades.page.html',
  styleUrls: ['./cidades.page.scss'],
  changeDetection: ChangeDetectionStrategy.OnPush,
  imports: [

    DestaqueAlertaDirective,
    DecimalPipe,
  ],
  providers: [DecimalPipe],
})
export class CidadesPage implements OnInit {
  cidades: CidadeSalva[] = [
    { nome: 'Recife, PE', lat: -8.0543, lon: -34.8813 },
    { nome: 'SÃ£o Paulo, SP', lat: -23.5505, lon: -46.6333 },
    { nome: 'Rio de Janeiro, RJ', lat: -22.9068, lon: -43.1729 },
    { nome: 'Curitiba, PR', lat: -25.4284, lon: -49.2733 },
    { nome: 'Fortaleza, CE', lat: -3.7172, lon: -38.5433 },
    { nome: 'Manaus, AM', lat: -3.1019, lon: -60.025 },
  ];

  carregandoGeral = true;

  // Busca e adiÃ§Ã£o de novas cidades
  termoBusca = '';
  buscando = false;
  sugestoes: ResultadoGeocodificacao[] = [];
  nenhumResultado = false;

  constructor(
    private climaService: ClimaService,
    private cdr: ChangeDetectorRef
  ) {}

  ngOnInit() {
    this.carregarTodasEmParalelo();
  }

  carregarTodasEmParalelo() {
    if (this.cidades.length === 0) {
      this.carregandoGeral = false;
      this.cdr.markForCheck();
      return;
    }

    this.carregandoGeral = true;
    this.cidades.forEach((c) => {
      c.carregando = true;
      c.erro = '';
    });

    const requisicoes = this.cidades.map((c) =>
      this.climaService.getClimaAtual(c.lat, c.lon)
    );

    forkJoin(requisicoes).subscribe({
      next: (resultados) => {
        resultados.forEach((data, i) => {
          this.cidades[i].clima = data;
          this.cidades[i].carregando = false;
        });
        this.carregandoGeral = false;
        this.cdr.markForCheck();
      },
      error: () => {
        this.cidades.forEach((c) => {
          c.erro = 'Erro ao carregar';
          c.carregando = false;
        });
        this.carregandoGeral = false;
        this.cdr.markForCheck();
      },
    });
  }

  // ðŸ” Pesquisa de cidades via API Geocoding
  onPesquisar(event: any) {
    const valor = event.target?.value?.trim() || '';
    this.termoBusca = valor;
    if (valor.length < 2) {
      this.sugestoes = [];
      this.nenhumResultado = false;
      this.cdr.markForCheck();
      return;
    }

    this.buscando = true;
    this.nenhumResultado = false;
    this.cdr.markForCheck();

    this.climaService.buscarCidades(valor).subscribe({
      next: (res) => {
        this.buscando = false;
        this.sugestoes = res.results || [];
        this.nenhumResultado = this.sugestoes.length === 0;
        this.cdr.markForCheck();
      },
      error: (err) => {
        console.error(err);
        this.buscando = false;
        this.sugestoes = [];
        this.cdr.markForCheck();
      },
    });
  }

  // âž• Adiciona a cidade pesquisada Ã  lista e busca o clima
  adicionarCidade(resultado: ResultadoGeocodificacao) {
    const nome = [resultado.name, resultado.admin1, resultado.country]
      .filter(Boolean)
      .join(', ');

    const novaCidade: CidadeSalva = {
      nome,
      lat: resultado.latitude,
      lon: resultado.longitude,
      carregando: true,
    };

    this.cidades.unshift(novaCidade);
    this.termoBusca = '';
    this.sugestoes = [];
    this.cdr.markForCheck();

    this.climaService.getClimaAtual(novaCidade.lat, novaCidade.lon).subscribe({
      next: (clima) => {
        novaCidade.clima = clima;
        novaCidade.carregando = false;
        this.cdr.markForCheck();
      },
      error: () => {
        novaCidade.erro = 'Erro ao consultar clima';
        novaCidade.carregando = false;
        this.cdr.markForCheck();
      },
    });
  }

  removerCidade(index: number) {
    this.cidades.splice(index, 1);
    this.cdr.markForCheck();
  }

  limparBusca() {
    this.termoBusca = '';
    this.sugestoes = [];
    this.nenhumResultado = false;
    this.cdr.markForCheck();
  }

  atualizarTodas() {
    this.climaService.limparCache();
    this.carregarTodasEmParalelo();
  }
}

    return `${lat.toFixed(4)}|${lon.toFixed(4)}`;
  }

  private cacheValido<T>(map: Map<string, { obs: Observable<T>; ts: number }>, key: string): Observable<T> | null {
    const entry = map.get(key);
    if (entry && Date.now() - entry.ts < this.CACHE_TTL_MS) {
      return entry.obs;

    return null;

@Injectable({
  providedIn: 'root'
})
export class HistoricoService {

  private chave = 'historico_cidades';

  // Adicionar cidade ao histÃ³rico
  adicionarCidade(cidade: string): void {
    let historico = this.obterHistorico();

    historico = historico.filter(
      item => item.toLowerCase() !== cidade.toLowerCase()
    );

    historico.unshift(cidade);

    historico = historico.slice(0, 5);

    localStorage.setItem(
      this.chave,
      JSON.stringify(historico)
    );
  }

  // Consultar histÃ³rico
  obterHistorico(): string[] {
    const dados = localStorage.getItem(this.chave);

    return dados ? JSON.parse(dados) : [];
  }

  // Remover uma cidade
  removerCidade(cidade: string): void {
    const historico = this.obterHistorico();

    const atualizado = historico.filter(
      item => item !== cidade
    );

    localStorage.setItem(
      this.chave,
      JSON.stringify(atualizado)
    );
  }

  // Limpar todo o histÃ³rico
  limparHistorico(): void {
    localStorage.removeItem(this.chave);
  }
}
